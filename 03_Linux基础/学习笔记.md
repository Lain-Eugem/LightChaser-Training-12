pwd                         # 显示当前目录的绝对路径
ls                          # 显示当前目录下存在的文件与文件夹（-l是更详细的信息；-a是显示隐藏文件）
cd (文件名)                 # 进入某一个文件(用/.表示当前目录，/..表示父目录)
mkdir (文件名)              # 建立一个叫()的空文件夹
touch (文件名)              # 建立一个空文件叫()
vim (文件名)                # 建立一个空文件并使用vim编辑器打开/直接使用vim编辑器打开叫做()的文件
cat (文件名)                # 打印()文件的内容到屏幕上
cp (源文件) (目标文件)       # 复制源文件到目标文件
rm (文件名)                 # 删除一个文件
rm -r (文件夹名)            # 递归删除一个文件夹及其内容（-r是递归的意思，谨慎使用）
mv (源文件) (目标文件)       # 移动源文件到目标文件（也可以用来重命名文件）
chmod (权限) (文件名)        # 改变文件的权限（权限可以是数字或符号）
./(文件名)                  # 执行当前目录下的()文件（前提是该文件有可执行权限）
sudo (命令)                   # 以超级用户身份执行命令（需要输入密码）
gcc (源文件) -o (目标文件)   # 编译源文件并生成可执行文件
echo $?                     # 显示上一个命令的退出状态码（0表示成功，非0表示失败）
